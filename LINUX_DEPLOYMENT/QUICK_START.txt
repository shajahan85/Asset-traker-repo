QUICK DEPLOYMENT (Copy & Paste)
═══════════════════════════════════════════════════════════════════════════════

# 1. Upload to server
scp -r LINUX_DEPLOYMENT username@server-ip:/tmp/

# 2. SSH to server
ssh username@server-ip

# 3. Backup current files
sudo mkdir -p /var/www/asset_tracker/backups/$(date +%Y%m%d_%H%M%S)
cd /var/www/asset_tracker
sudo cp -r assets/forms/ backups/$(date +%Y%m%d_%H%M%S)/
sudo cp -r assets/views*.py backups/$(date +%Y%m%d_%H%M%S)/
sudo cp -r assets/services/ backups/$(date +%Y%m%d_%H%M%S)/
sudo cp -r assets/templates/ backups/$(date +%Y%m%d_%H%M%S)/

# 4. Deploy new files
cd /tmp/LINUX_DEPLOYMENT
sudo cp -r assets/* /var/www/asset_tracker/assets/

# 5. Set permissions
sudo chown -R www-data:www-data /var/www/asset_tracker/assets
sudo chmod -R 755 /var/www/asset_tracker/assets

# 6. Collect static files
cd /var/www/asset_tracker
source venv/bin/activate
python manage.py collectstatic --noinput

# 7. Restart service
sudo systemctl restart asset_tracker
sudo systemctl status asset_tracker
